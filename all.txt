import java.sql.*;
class j6{
    public static void main(String args[]) {
        try {
            Class.forName("oracle.jdbc.driver.OracleDriver");
            Connection con = DriverManager.getConnection(
            "jdbc:oracle:thin:@localhost:1521:xe","system","password"
            );
            Statement st=con.createStatement();
            st.executeUpdate("create table stud(rno number, name varchar2(20))");
            st.executeUpdate("insert into stud values(101, 'Alice')");
            ResultSet rs=st.executeQuery("select * from stud");

                System.out.println(rs.getInt(1));
                System.out.println(rs.getString(2));
            }
            st.close();
            con.close();
}catch(Exception e){

//7)Prepared Statement

import java.sql.*;
class j7 {
    public static void main(String args[]) {
        try {
            Class.forName("oracle.jdbc.driver.OracleDriver");
            Connection con = DriverManager.getConnection(
            "jdbc:oracle:thin:@localhost:1521:xe","system","password"
            );
            Statement st = con.createStatement();
            st.executeUpdate("create table stud(rno number, name varchar2(20))");
            PreparedStatement ps = con.prepareStatement("insert into stud values(?, ?)");
            ps.setInt(1, 101); 
            ps.setString(2, "Alice");
            ps.executeUpdate();
            System.out.println("Record inserted successfully using PreparedStatement!");
            ps.close();
            con.close();
}catch(Exception e){
e.printStackTrace();
}}}

//8)callable statement

import java.sql.*;
class j8{
    public static void main(String args[]) {
        try {
            Class.forName("oracle.jdbc.driver.OracleDriver");
            Connection con = DriverManager.getConnection(
            "jdbc:oracle:thin:@localhost:1521:xe","system","password"
            );
            CallableStatement cs=con.prepareCall("{call procedurename(?,?)}");
            cs.setInt(1, 101);
            cs.setString(2, "Alice");
            cs.execute();
}catch(Exception e){
e.printStackTrace();
}}}

SQL PROCEDURE CREATION:

CREATE OR REPLACE PROCEDURE procedurename
(rno NUMBER, name VARCHAR2)
AS
BEGIN
   INSERT INTO tablename VALUES (rno, name);
END;

import java.io.*;
import java.util.Date;
import javax.servlet.*;
import javax.servlet.http.*;

public class DemoSession extends HttpServlet {

    public void doGet(HttpServletRequest req, HttpServletResponse res)
            throws ServletException, IOException {

        HttpSession session = req.getSession();

        res.setContentType("text/html");
        PrintWriter pw = res.getWriter();

        pw.println("Welcome arisu");
        pw.println("Session ID: " + session.getId());
        pw.println("Creation Time: " + new Date(session.getCreationTime()));
        pw.println("Last Accessed Time: " + new Date(session.getLastAccessedTime()));
    }

//14)Servlet Program To Demonstrate Cookies
import java.io.*;
import javax.servlet.*;
import javax.servlet.http.*;

public class setcookie extends HttpServlet {
    public void doGet(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {
        Cookie cookie = new Cookie("username", "arisu");
        cookie.setMaxAge(60 * 60);
        response.addCookie(cookie);
        PrintWriter out = response.getWriter();
        out.println("Cookie has been set!");

//SerReqFwd.java
import javax.servlet.*;
import javax.servlet.http.*;
import java.io.*;

public class SerReqFwd extends HttpServlet {

public void doPost(HttpServletRequest req, HttpServletResponse res)
throws ServletException, IOException {

res.setContentType("text/html");
PrintWriter pw = res.getWriter();

String a = req.getParameter("a1");
String b = req.getParameter("b1");

int no1 = Integer.parseInt(a);
int no2 = Integer.parseInt(b);

int sum = no1 + no2;

req.setAttribute("sum", sum);

RequestDispatcher rd = req.getRequestDispatcher("/Servlet2");
rd.forward(req, res);
}}
