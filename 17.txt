//17) Implicit objects in JSP

//implicit.jsp
<html>
<head>
<title>Implicit objects</title>
</head>
<body>
<%
String name1 = request.getParameter("a1");
String pwd1 = request.getParameter("a2");

out.println("Name: " + name1 + "<br>");
out.println("Password: " + pwd1);
%>
</body>
</html>


//Login.html
<html>
<head>
</head>
<body>
<form action="implicit.jsp" method="post">

<label>Name</label>
<input type="text" name="a1">

<label>Password</label>
<input type="password" name="a2">

<input type="submit" value="login">
</form>
</body>
</html>
