//20) jsp program to demonstrate jstl 

<%@ page language="java" contentType="text/html; charset=UTF-8" %>
<%@ taglib uri="http://java.sun.com/jsp/jstl/core" prefix="c" %>

<html>
<head>
    <title>JSTL Core Tags Demo</title>
</head>
<body>

<c:set var="name" value="Izuki" />
<c:set var="marks" value="85" />

<h3>Name: <c:out value="${name}" /></h3>
<h3>Marks: <c:out value="${marks}" /></h3>

<c:if test="${marks >= 40}">
    <p>Status: Pass</p>
</c:if>

<h4>Subjects:</h4>

<ul>
    <c:forEach var="subject" items="${['Maths','Science','English']}">
        <li><c:out value="${subject}" /></li>
    </c:forEach>
</ul>

</body>
</html>
